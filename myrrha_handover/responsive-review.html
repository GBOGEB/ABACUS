Orientation CSS

Recommended responsive rules
/* Portrait mobile (9:16) */ @media (max-width: 768px) and (orientation: portrait) { .chart { width: 100%; height: auto; } .sidebar { display: none; } /* Collapsible hamburger */ } /* Landscape mobile (16:9) */ @media (max-width: 768px) and (orientation: landscape) { .chart { width: 50%; float: left; } .sidebar { width: 30%; } /* Utilise horizontal space */ } /* Desktop (16:9) */ @media (min-width: 1024px) { .chart { width: 60%; margin: 0 auto; } /* Centre, avoid full-width stretch */ .sidebar { width: 20%; position: sticky; } /* Side panels for metadata */ }
